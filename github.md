# Git Command to push the code

1. git init
2. git status
3. git add .
4. git status
5. git commit -m "my first commit"
6. git remote add origin https://github.com/USERNAME/my-project.git
7. git branch
8. git branch -M main
9. git push -u origin main


# After updation in code

1. git status
2. git add .
3. git commit -m "Updated project"
4. git push
